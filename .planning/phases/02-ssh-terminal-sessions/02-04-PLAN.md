**Create MobaAlt/Views/Terminal/TerminalTabBar.swift:**

A custom SwiftUI view using `ScrollView(.horizontal, showsIndicators: false)` + `HStack`. Each tab item shows the session name, an active indicator, and an "xmark" close button.

```swift
struct TerminalTabBar: View {
    @Bindable var tabManager: TabManager

    var body: some View {
        ScrollView(.horizontal, showsIndicators: false) {
            HStack(spacing: 0) {
                ForEach(tabManager.tabs) { tab in
                    TabBarItem(
                        tab: tab,
                        isActive: tab.id == tabManager.activeTabId,
                        onSelect: { tabManager.activateTab(tab.id) },
                        onClose:  { tabManager.closeTab(tab.id) }
                    )
                }
            }
        }
        .frame(height: 32)
        .background(.regularMaterial)
    }
}

struct TabBarItem: View {
    let tab: TabItem
    let isActive: Bool
    let onSelect: () -> Void
    let onClose: () -> Void
    // Background: active = .regularMaterial; inactive = .clear
    // Text: tab.displayName truncated to 20 chars, foreground secondary when inactive
    // Close button: Image(systemName: "xmark").font(.system(size: 9))
}
```

**Create MobaAlt/Views/Terminal/TerminalTabView.swift:**

`NSViewRepresentable` wrapping `SSHConnection.terminalView`. The coordinator is SSHConnection's Coordinator (not the connection itself — keep view and logic separate).

```swift
struct TerminalTabView: NSViewRepresentable {
    let connection: SSHConnection

    func makeNSView(context: Context) -> NSView {
        // connection.terminalView is set by SSHConnection.start() on @MainActor
        // If terminalView is nil (connection not started yet), return a placeholder NSView
        return connection.terminalView ?? NSView()
    }

    func updateNSView(_ nsView: NSView, context: Context) {
        // Terminal manages its own rendering — no-op
    }
}
```

NOTE: Because `connection.start()` must be called to create `terminalView`, call `connection.start()` in `makeNSView` if `connection.terminalView == nil`. Use `Task { await connection.start() }` if @MainActor is needed. This ensures the terminal is started exactly once.
