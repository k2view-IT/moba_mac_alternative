    SFTPFileListView.swift:
    - Struct SFTPFileListView: View. Takes service: SFTPBrowserService, sortColumn: Binding<SFTPSortColumn>, sortAscending: Binding<Bool>.
    - @State private var selection: Set<String> = [] (Set of SFTPItem.id/path for multi-select)
    - @State private var renamingItemId: String? = nil (which item is in inline-rename mode)
    - @State private var renameText: String = ""

    Compute sorted items:
    ```swift
    private var sortedItems: [SFTPItem] {
        let sorted: [SFTPItem]
        switch sortColumn.wrappedValue {
        case .name:
            sorted = service.items.sorted {
                if $0.isDirectory != $1.isDirectory { return $0.isDirectory }  // folders first
                return sortAscending.wrappedValue ? $0.name < $1.name : $0.name > $1.name
            }
        case .size:
            sorted = service.items.sorted { sortAscending.wrappedValue ? $0.size < $1.size : $0.size > $1.size }
        case .date:
            sorted = service.items.sorted { sortAscending.wrappedValue ? $0.modificationDate < $1.modificationDate : $0.modificationDate > $1.modificationDate }
        }
        return sorted
    }
    ```

    Column header row: HStack with three clickable Text buttons. Clicking the same column toggles sortAscending; clicking a different column resets to ascending. Show chevron.up/chevron.down SF Symbol next to active column.

    List body: List(sortedItems, selection: $selection) { item in SFTPFileRowView(item: item, service: service, isRenaming: ...) }.onChange(of: selection) { ... }
