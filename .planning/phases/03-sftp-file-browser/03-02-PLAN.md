    - init(socketPath: String, sessionId: UUID, config: SSHConfig) throws: spawns /usr/bin/sftp with arguments: ["-b", "-", "-o", "ControlMaster=no", "-o", "ControlPath=\(socketPath)", "\(config.username)@\(config.hostname)"]. Sets up stdin/stdout/stderr Pipes.
    - connect(): calls process.run(), reads sftp startup banner from stdout (waits for "sftp>" prompt), confirms connection ready.
    - send(_ command: String): writes command + "\n" to stdinPipe.fileHandleForWriting.
    - readOutput() -> String: reads available data from stdoutPipe.fileHandleForReading (non-blocking, up to prompt).
